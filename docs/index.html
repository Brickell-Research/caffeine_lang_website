Caffeine

A compiler for generating reliability artifacts from service expectation definitions.

Get StartedRead the Blog

Installation

With Homebrew:

brew tap Brickell-Research/caffeine_lang && brew install caffeine
Specification
basic_types:
  - attribute_name: gql_operation
    attribute_type: String
  - attribute_name: numerator
    attribute_type: String
  - attribute_name: denominator
    attribute_type: String

query_template_types:
  - name: "valid_over_total"
    specification_of_query_templates: ["numerator", "denominator"]
    query: "numerator / denominator"
types:
  - name: success_rate
    query_template_type: valid_over_total
    typed_instatiation_of_query_templates:
      numerator: "sum.graphql.requests{$$operation->gql_operation$$, status:info}.as_count()"
      denominator: "sum.graphql.requests{$$operation->gql_operation$$}.as_count()"
    specification_of_query_templatized_variables:
      - gql_operation
services:
  - name: reliable_service
    sli_types:
      - success_rate
